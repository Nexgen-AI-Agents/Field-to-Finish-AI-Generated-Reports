DATA QUERIES USED

-- Monthly volume by year
SELECT MONTH(ng_date) as month, YEAR(ng_date) as year, COUNT(*) as total_orders
FROM ng_orders
WHERE YEAR(ng_date) IN (2025, 2026) AND MONTH(ng_date) BETWEEN 1 AND 4
GROUP BY YEAR(ng_date), MONTH(ng_date) ORDER BY YEAR(ng_date), MONTH(ng_date);

-- Monthly completions/cancels/go-backs by year
SELECT MONTH(ng_date) as month, YEAR(ng_date) as year,
       SUM(CASE WHEN ng_status IN (7,8) THEN 1 ELSE 0 END) as completed,
       SUM(CASE WHEN ng_status = 0 THEN 1 ELSE 0 END) as cancelled,
       SUM(CASE WHEN is_go_back = 1 THEN 1 ELSE 0 END) as go_backs
FROM ng_orders
WHERE YEAR(ng_date) IN (2025, 2026) AND MONTH(ng_date) BETWEEN 1 AND 4
GROUP BY YEAR(ng_date), MONTH(ng_date) ORDER BY YEAR(ng_date), MONTH(ng_date);

-- Crew go-back rates by year
SELECT ng_crew, YEAR(ng_date) as year,
       COUNT(*) as total,
       SUM(CASE WHEN is_go_back=1 THEN 1 ELSE 0 END) as go_backs,
       ROUND(SUM(CASE WHEN is_go_back=1 THEN 1 ELSE 0 END)/COUNT(*)*100,1) as gb_pct
FROM ng_orders
WHERE YEAR(ng_date) IN (2025, 2026) AND MONTH(ng_date) BETWEEN 1 AND 4
  AND ng_crew IS NOT NULL AND ng_crew != ''
GROUP BY ng_crew, YEAR(ng_date) ORDER BY ng_crew, YEAR(ng_date);

-- Drafter go-back rates by year
SELECT ng_drafter, YEAR(ng_date) as year,
       COUNT(*) as total,
       SUM(CASE WHEN is_go_back=1 THEN 1 ELSE 0 END) as go_backs,
       ROUND(SUM(CASE WHEN is_go_back=1 THEN 1 ELSE 0 END)/COUNT(*)*100,1) as gb_pct
FROM ng_orders
WHERE YEAR(ng_date) IN (2025, 2026) AND MONTH(ng_date) BETWEEN 1 AND 4
  AND ng_drafter IS NOT NULL AND ng_drafter != ''
GROUP BY ng_drafter, YEAR(ng_date) ORDER BY ng_drafter, YEAR(ng_date);

-- Coordinator performance
SELECT survey_coordinator, YEAR(ng_date) as year, COUNT(*) as total,
       SUM(CASE WHEN ng_status IN (7,8) THEN 1 ELSE 0 END) as completed,
       SUM(CASE WHEN ng_status = 0 THEN 1 ELSE 0 END) as cancelled,
       SUM(CASE WHEN is_go_back = 1 THEN 1 ELSE 0 END) as go_backs
FROM ng_orders
WHERE YEAR(ng_date) IN (2025, 2026) AND MONTH(ng_date) BETWEEN 1 AND 4
  AND survey_coordinator IS NOT NULL AND survey_coordinator != ''
GROUP BY survey_coordinator, YEAR(ng_date) ORDER BY survey_coordinator, YEAR(ng_date);
