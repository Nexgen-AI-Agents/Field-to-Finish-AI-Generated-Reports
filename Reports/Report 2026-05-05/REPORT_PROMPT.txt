DATA QUERIES USED
──────────────────
The following MySQL queries were used to gather data for this report.
All queries run against nexgen_ftf_db via the mcp_server_mysql MCP server.

-- Total orders and status summary (2026)
SELECT ng_status, COUNT(*) as cnt
FROM ng_orders
WHERE YEAR(order_date) = 2026
GROUP BY ng_status;

-- Monthly order volume
SELECT DATE_FORMAT(order_date,'%Y-%m') as month, COUNT(*) as orders
FROM ng_orders
WHERE YEAR(order_date) = 2026
GROUP BY month ORDER BY month;

-- Crew go-back rates
SELECT ng_crew,
       COUNT(*) as total,
       SUM(CASE WHEN is_go_back=1 THEN 1 ELSE 0 END) as go_backs,
       ROUND(SUM(CASE WHEN is_go_back=1 THEN 1 ELSE 0 END)/COUNT(*)*100,1) as gb_pct
FROM ng_orders
WHERE YEAR(order_date) = 2026 AND ng_crew IS NOT NULL AND ng_crew != ''
GROUP BY ng_crew ORDER BY total DESC;

-- Status transition timing (pipeline bottlenecks)
SELECT from_status, to_status,
       ROUND(AVG(DATEDIFF(changed_at, prev_changed_at)),2) as avg_days,
       COUNT(*) as transitions,
       SUM(CASE WHEN DATEDIFF(changed_at, prev_changed_at) > 7 THEN 1 ELSE 0 END) as over_7_days
FROM (
  SELECT ng_order_id, ng_status as to_status,
         LAG(ng_status) OVER (PARTITION BY ng_order_id ORDER BY changed_at) as from_status,
         changed_at,
         LAG(changed_at) OVER (PARTITION BY ng_order_id ORDER BY changed_at) as prev_changed_at
  FROM order_status_tracking
  WHERE YEAR(changed_at) = 2026
) t
WHERE from_status IS NOT NULL
GROUP BY from_status, to_status
ORDER BY avg_days DESC;

-- Open orders (saved to open_orders_2026.json)
SELECT ng_order, ng_property_address, ng_property_city, ng_property_county,
       ng_status, ng_crew, ng_drafter, survey_coordinator,
       order_date, fieldwork_date, delivered_date, last_status_change,
       DATEDIFF(NOW(), last_status_change) as days_in_current_status,
       DATEDIFF(NOW(), order_date) as total_age_days,
       is_go_back, ng_access_call_needed, ng_access_call_settled, ng_company_id
FROM ng_orders
WHERE YEAR(order_date) = 2026
  AND ng_status NOT IN (0, 7, 8)
ORDER BY days_in_current_status DESC;

-- County performance
SELECT ng_property_county,
       COUNT(*) as total,
       SUM(CASE WHEN ng_status IN (7,8) THEN 1 ELSE 0 END) as completed,
       SUM(CASE WHEN ng_status = 0 THEN 1 ELSE 0 END) as cancelled,
       SUM(CASE WHEN is_go_back = 1 THEN 1 ELSE 0 END) as go_backs
FROM ng_orders
WHERE YEAR(order_date) = 2026
GROUP BY ng_property_county
ORDER BY total DESC;
