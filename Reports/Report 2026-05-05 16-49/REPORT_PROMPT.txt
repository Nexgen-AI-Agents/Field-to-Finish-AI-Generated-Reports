DATA QUERIES USED
──────────────────
The following MySQL queries were used to gather data for this report.
All queries run against nexgen_ftf_db via the mcp_server_mysql MCP server.
NOTE: Date field is ng_date (not order_date).

-- Total orders and status summary (2025)
SELECT ng_status, COUNT(*) as cnt
FROM ng_orders
WHERE YEAR(ng_date) = 2025
GROUP BY ng_status;

-- Monthly order volume
SELECT DATE_FORMAT(ng_date,'%Y-%m') as month, COUNT(*) as orders
FROM ng_orders
WHERE YEAR(ng_date) = 2025
GROUP BY month ORDER BY month;

-- Crew go-back rates
SELECT ng_crew,
       COUNT(*) as total,
       SUM(CASE WHEN is_go_back=1 THEN 1 ELSE 0 END) as go_backs,
       ROUND(SUM(CASE WHEN is_go_back=1 THEN 1 ELSE 0 END)/COUNT(*)*100,1) as gb_pct
FROM ng_orders
WHERE YEAR(ng_date) = 2025 AND ng_crew IS NOT NULL AND ng_crew != ''
GROUP BY ng_crew ORDER BY total DESC;

-- Status transition timing (pipeline bottlenecks)
-- Note: order_status_tracking uses old_status, new_status, duration_seconds, started_at, end_at
SELECT old_status, new_status,
       ROUND(AVG(duration_seconds/86400.0),2) as avg_days,
       COUNT(*) as transitions,
       SUM(CASE WHEN duration_seconds > 604800 THEN 1 ELSE 0 END) as over_7_days
FROM order_status_tracking
WHERE YEAR(started_at) = 2025
  AND old_status IS NOT NULL
GROUP BY old_status, new_status
ORDER BY avg_days DESC;

-- Open orders at year-end 2025
SELECT ng_order, ng_property_address, ng_property_city, ng_property_county,
       ng_status, ng_crew, ng_drafter, survey_coordinator,
       ng_date, fieldwork_date, delivered_date, last_status_change,
       DATEDIFF(NOW(), last_status_change) as days_in_current_status,
       DATEDIFF(NOW(), ng_date) as total_age_days,
       is_go_back, ng_access_call_needed, ng_access_call_settled, ng_company_id
FROM ng_orders
WHERE YEAR(ng_date) = 2025
  AND ng_status NOT IN (0, 7, 8)
ORDER BY days_in_current_status DESC;

-- County performance
SELECT ng_property_county,
       COUNT(*) as total,
       SUM(CASE WHEN ng_status IN (7,8) THEN 1 ELSE 0 END) as completed,
       SUM(CASE WHEN ng_status = 0 THEN 1 ELSE 0 END) as cancelled,
       SUM(CASE WHEN is_go_back = 1 THEN 1 ELSE 0 END) as go_backs
FROM ng_orders
WHERE YEAR(ng_date) = 2025
GROUP BY ng_property_county
ORDER BY total DESC;

-- Access call backlog
SELECT COUNT(*) as total_needing_access,
       SUM(CASE WHEN ng_access_call_settled = 0 OR ng_access_call_settled IS NULL THEN 1 ELSE 0 END) as unsettled
FROM ng_orders
WHERE YEAR(ng_date) = 2025
  AND ng_access_call_needed = 1;
